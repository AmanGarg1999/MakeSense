"""Background job queue demo app (stdlib only)."""
import json, sqlite3, threading, time, traceback
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer

DB = "jobs.db"

def conn():
    c = sqlite3.connect(DB, timeout=10)
    c.execute("""CREATE TABLE IF NOT EXISTS jobs(
        id INTEGER PRIMARY KEY, payload TEXT, status TEXT DEFAULT 'queued',
        result TEXT, tries INTEGER DEFAULT 0, ts REAL)""")
    return c

def worker(stop):
    while not stop.is_set():
        c = conn()
        try:
            row = c.execute("SELECT id,payload,tries FROM jobs WHERE status='queued' ORDER BY id LIMIT 1").fetchone()
            if row:
                jid, payload, tries = row
                c.execute("UPDATE jobs SET status='running' WHERE id=?", (jid,))
                c.commit()
                try:
                    out = process(json.loads(payload))
                    c.execute("UPDATE jobs SET status='done', result=? WHERE id=?", (json.dumps(out), jid))
                except Exception as e:
                    # retry with capped backoff
                    if tries < 4:
                        c.execute("UPDATE jobs SET status='queued', tries=tries+1 WHERE id=?", (jid,))
                    else:
                        c.execute("UPDATE jobs SET status='failed', result=? WHERE id=?", (str(e), jid))
                c.commit()
        finally:
            c.close()
        time.sleep(0.5)

def process(job):
    kind = job["kind"]
    if kind == "resize":
        # simulate work
        time.sleep(0.1)
        return {"ok": True, "w": job["w"], "h": job["h"]}
    if kind == "report":
        rows = conn().execute("SELECT status, COUNT(*) FROM jobs GROUP BY status").fetchall()
        return {"ok": True, "counts": dict(rows)}
    raise ValueError(f"unknown kind {kind}")

class H(BaseHTTPRequestHandler):
    def do_POST(self):
        body = self.rfile.read(int(self.headers["Content-Length"]))
        try:
            payload = json.loads(body)
            assert "kind" in payload
        except Exception:
            self.send_response(400); self.end_headers(); return
        c = conn()
        c.execute("INSERT INTO jobs(payload,ts) VALUES(?,?)", (json.dumps(payload), time.time()))
        c.commit(); jid = c.execute("SELECT max(id) FROM jobs").fetchone()[0]; c.close()
        self.send_response(202)
        self.send_header("Content-Type", "application/json"); self.end_headers()
        self.wfile.write(json.dumps({"job_id": jid}).encode())

    def do_GET(self):
        if self.path.startswith("/jobs/"):
            jid = self.path.split("/")[-1]
            r = conn().execute("SELECT status,result FROM jobs WHERE id=?", (jid,)).fetchone()
            if not r:
                self.send_response(404); self.end_headers(); return
            self.send_response(200)
            self.send_header("Content-Type", "application/json"); self.end_headers()
            self.wfile.write(json.dumps({"status": r[0], "result": r[1]}).encode())
        else:
            self.send_response(200); self.send_header("Content-Type", "text/html"); self.end_headers()
            self.wfile.write(b"<html><body><h1>Job Queue</h1><form onsubmit='fetch(\"/jobs\",{method:\"POST\",body:JSON.stringify({kind:this.k.value})});return false'><input name=k value=resize><button>enqueue</button></form></body></html>")

def main():
    stop = threading.Event()
    t = threading.Thread(target=worker, args=(stop,), daemon=True)
    t.start()
    srv = ThreadingHTTPServer(("0.0.0.0", 8000), H)
    try:
        srv.serve_forever()
    finally:
        stop.set()

if __name__ == "__main__":
    main()
